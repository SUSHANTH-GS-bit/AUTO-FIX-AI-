import 'dart:convert'; import 'package:http/http.dart' as http; import '../models/car_model.dart'; class ClaudeService { // IMPORTANT: Never put API key directly here in production! // Use Firebase Remote Config or environment variables instead. static const _apiKey = 'YOUR_CLAUDE_API_KEY_HERE'; static const _baseUrl = 'https://api.anthropic.com/v1/messages'; static String _buildSystemPrompt(CarModel? car) { final carInfo = car != null ? 'The user\'s car is: ${car.displayName}.' : 'The user has not selected a car yet.'; return '''You are Auto Fix AI — a friendly expert car mechanic for Indian car owners. $carInfo Your rules: 1. Answer in simple, clear language — no technical jargon 2. Keep answers under 5 sentences unless detailed guide needed 3. Always mention estimated cost in Indian Rupees (₹) 4. If the problem is serious or dangerous, say: "Visit a mechanic immediately — do not drive this car" 5. If user writes in Hindi or Kannada, reply in that language 6. Always give one practical first step the user can do now 7. Be warm and reassuring — users are stressed about their car'''; } static Future<String> diagnose({ required String userMessage, CarModel? car, }) async { try { final response = await http.post( Uri.parse(_baseUrl), headers: { 'Content-Type': 'application/json', 'x-api-key': _apiKey, 'anthropic-version': '2023-06-01', }, body: jsonEncode({ 'model': 'claude-sonnet-4-20250514', 'max_tokens': 500, 'system': _buildSystemPrompt(car), 'messages': [ {'role': 'user', 'content': userMessage} ], }), ); if (response.statusCode == 200) { final data = jsonDecode(response.body); return data['content'][0]['text'] as String; } else { // Fallback to offline answer if API fails return _getOfflineAnswer(userMessage); } } catch (e) { // No internet — use offline database return _getOfflineAnswer(userMessage); } } // Offline fallback answers for common problems static String _getOfflineAnswer(String problem) { final p = problem.toLowerCase(); if (p.contains('engine light') || p.contains('check engine')) { return '⚠️ Engine warning light usually means a sensor issue. ' 'Common causes: loose fuel cap (₹0 fix — tighten it), ' 'oxygen sensor fault (₹1,500–3,000), or catalytic converter. ' 'First step: tighten your fuel cap and restart the car. ' 'If light stays on after 2 trips, visit a mechanic.'; } if (p.contains('ac') || p.contains('cooling')) { return '❄️ AC not cooling is usually low refrigerant gas. ' 'Estimated cost: AC gas refill ₹800–1,500 at any garage. ' 'Also check: cabin air filter (₹200–400 to replace). ' 'First step: check if AC compressor is running when AC is ON.'; } if (p.contains('battery')) { return '🔋 Battery warning means charging system issue. ' 'Could be: battery needs replacement (₹3,000–5,000), ' 'or alternator fault (₹4,000–8,000). ' 'First step: check battery terminals for corrosion — ' 'clean white powder with a dry cloth if you see it.'; } if (p.contains('brake') || p.contains('squeak')) { return '🛑 Brake squeaking usually means worn brake pads. ' 'Replacement cost: ₹1,200–2,500 for front pads. ' 'This is important for safety — do not ignore it. ' 'First step: check if squeaking happens when you press ' 'OR release the brake pedal — tell the mechanic which one.'; } return '🔧 I need internet to give you a detailed answer for this. ' 'Please connect to WiFi or mobile data and ask again. ' 'For urgent help, call a trusted mechanic directly.'; } }

import 'package:cloud_firestore/cloud_firestore.dart'; import 'package:firebase_auth/firebase_auth.dart'; import '../models/car_model.dart'; class FirebaseService { static final _db = FirebaseFirestore.instance; static final _auth = FirebaseAuth.instance; static String? get userId => _auth.currentUser?.uid; // Save user's car to Firebase static Future<void> saveCar(CarModel car) async { if (userId == null) return; await _db.collection('users').doc(userId).set( {'car': car.toMap()}, SetOptions(merge: true), ); } // Get user's saved car static Future<CarModel?> getSavedCar() async { if (userId == null) return null; final doc = await _db.collection('users').doc(userId).get(); final data = doc.data()?['car']; if (data == null) return null; return CarModel.fromMap(data as Map<String, dynamic>); } // Get nearby garages (sorted by city for now) static Future<List<GarageModel>> getGarages(String city) async { final snap = await _db.collection('garages') .where('city', isEqualTo: city) .where('isOpen', isEqualTo: true) .limit(20) .get(); return snap.docs .map((d) => GarageModel.fromMap({...d.data(), 'id': d.id})) .toList(); } // Get spare parts filtered by car brand static Future<List<Map<String,dynamic>>> getParts( {String? carBrand, String? search}) async { Query q = _db.collection('parts').limit(50); if (carBrand != null && carBrand.isNotEmpty) { q = q.where('carBrand', whereIn: [carBrand, 'Universal']); } final snap = await q.get(); var parts = snap.docs .map((d) => {...d.data() as Map, 'id': d.id}) .toList(); // Filter by search query locally if (search != null && search.isNotEmpty) { final s = search.toLowerCase(); parts = parts.where((p) => (p['name'] as String).toLowerCase().contains(s) ).toList(); } return parts; } // Save a chat message to history static Future<void> saveChat( String userMsg, String aiReply) async { if (userId == null) return; await _db.collection('chats').add({ 'userId': userId, 'userMessage': userMsg, 'aiReply': aiReply, 'timestamp': FieldValue.serverTimestamp(), }); } }
